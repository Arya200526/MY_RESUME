import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

df = pd.read_csv(r'C:\Users\aryag\OneDrive\Apps\Attachments\Desktop\Air and Heart\Air_Quality.csv',delimiter=',')

df.head()

df.shape
df.info()
df.describe()
df.isnull().sum()


#Corrleation Heatmap
plt.figure(figsize=(10,5))    
sns.heatmap(df.corr(numeric_only=True), annot=True, cmap='coolwarm',fmt=".2f")
plt.title('Correlation Heatmap')
plt.show()

#Line plot
plt.figure(figsize=(7,8))
sns.lineplot(data=df, x='Start_Date', y='Data Value')
plt.show()
plt.title('Start_Date vs Data Value')
plt.xlabel('Start_Date')
plt.ylabel('Data Value')
plt.show()

#Bar plot
plt.figure(figsize=(9,6))
sns.barplot(data=df, x='Geo Place Name', y='Data Value')
plt.show()
plt.title('Geo Place Name vs Data Value')
plt.xlabel('Geo Place Name')
plt.ylabel('Data Value')
plt.show()

#Scatter plot
plt.figure(figsize=(6,7))
sns.scatterplot(data=df, x='Time Period', y='Data Value')
plt.show()
plt.title('Time Period vs Data Value')
plt.xlabel('Time Period')
plt.ylabel('Data Value')
plt.show()

#pie chart
plt.figure(figsize=(7,9))
sns.boxplot(data=df, x='Geo Place Name',y='Data Value')
plt.show()
plt.title('Geo Place Name vs Data Value')
plt.xlabel('Geo Place Name')
plt.ylabel('Data Value')
plt.show()
